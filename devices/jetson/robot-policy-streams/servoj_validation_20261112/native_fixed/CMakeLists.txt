cmake_minimum_required(VERSION 3.16)
project(robot_timed_policy_driver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rokae_interfaces REQUIRED)
find_package(Threads REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
set(ROBOT_SOURCE_ROOT "/home/jetson/AIDE_robot_main" CACHE PATH "Existing robot SDK")
set(SDK_ROOT "${ROBOT_SOURCE_ROOT}/hardware/arm/xCoreSDK-v0.7.1.ar_6")
set(SDK_LIB_DIR "${SDK_ROOT}/lib/Linux/aarch64")
add_executable(robot_timed_policy_driver main.cpp
  vendor/src/ros_robot_initializer_service.cpp vendor/src/ros_moveabsj_service.cpp
  vendor/src/ros_servoj_subscriber.cpp)
target_compile_definitions(robot_timed_policy_driver PRIVATE ROKAE_UNIFIED_DRIVER)
target_compile_options(robot_timed_policy_driver PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(robot_timed_policy_driver PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include"
  "${SDK_ROOT}/include" "${SDK_ROOT}/external")
target_link_libraries(robot_timed_policy_driver
  "${SDK_LIB_DIR}/libxCoreSDK.a" "${SDK_LIB_DIR}/libxMateModel.a" Threads::Threads)
ament_target_dependencies(robot_timed_policy_driver rclcpp std_msgs std_srvs sensor_msgs geometry_msgs rokae_interfaces tf2 tf2_geometry_msgs)
add_executable(test_timed_policy test_timed_policy.cpp)
target_compile_options(test_timed_policy PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_executable(test_latest_feedback test_latest_feedback.cpp)
target_compile_options(test_latest_feedback PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_executable(readonly_feedback_probe readonly_feedback_probe.cpp)
target_include_directories(readonly_feedback_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SDK_ROOT}/include" "${SDK_ROOT}/external")
target_link_libraries(readonly_feedback_probe "${SDK_LIB_DIR}/libxCoreSDK.a" "${SDK_LIB_DIR}/libxMateModel.a" Threads::Threads)
