cmake_minimum_required(VERSION 3.12)
project(ar5_measurement LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|x86_64)$")
  message(FATAL_ERROR "Unsupported SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(SDK_LIB "${SDK_ROOT}/lib/Linux/${CMAKE_SYSTEM_PROCESSOR}")
add_executable(inspect_ar5_sdk inspect_ar5_sdk.cpp)
target_include_directories(inspect_ar5_sdk PRIVATE "${SDK_ROOT}/include" "${SDK_ROOT}/external")
target_link_libraries(inspect_ar5_sdk PRIVATE "${SDK_LIB}/libxCoreSDK.a" "${SDK_LIB}/libxMateModel.a" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(inspect_ar5_sdk PRIVATE -Wall -Wextra -Wpedantic)
